// Fisher-Yates Shuffle Algorithm

function shuffleArray(array) {

    let shuffled = [...array];

    for (let i = shuffled.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];

    }

    return shuffled;

}

        }


        ***** ********* = document.createElement('div');

            <div class="explanation"><b>Explanation:</b> ${q.explanation}</div>

        `;

        reviewContainer.appendChild(reviewDiv);

    });

    

    ***** total = rawQuestionsDB.length;

    const accuracy = correct + wrong > 0 ? Math.round((correct / (correct + wrong)) * 100) : 0;

    

    document.getElementById('r-score').innerText = `${correct}/${total}`;

    document.getElementById('r-percent').innerText = `${Math.round((correct/total)*100)}%`;

    document.getElementById('r-accuracy').innerText = `${accuracy}%`;

    document.getElementById('r-time').innerText = `${Math.floor(timeUsed/60)}m ${timeUsed%60}s`;

    

    renderCharts(correct, wrong, unanswered, sectionData);

}
